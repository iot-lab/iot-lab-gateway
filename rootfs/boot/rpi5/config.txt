# Raspberry Pi 5 gateway: firmware configuration for the EEPROM netboot,
# loaded from the per-node TFTP directory of the site TFTP server.

# Enable UART on the 40-pin header (RP1's uart0, GPIO14/15). The Pi5's
# device-tree default console is uart10, the board's separate 3-pin debug
# UART connector
# uart_2ndstage surfaces the EEPROM's own 2nd-stage bootloader messages here
# too, before Linux even starts.
enable_uart=1
uart_2ndstage=1
dtparam=uart0_console

# Headless NFS-root gateway, no need for the onboard Bluetooth chip
dtoverlay=disable-bt

# Kernel file name in the TFTP directory (default: kernel_2712.img)
kernel=Image
